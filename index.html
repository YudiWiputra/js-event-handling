1. Inline Event Handling Dasar

Event handler ditulis langsung di atribut HTML. Cara paling sederhana, namun tidak disarankan untuk proyek besar karena mencampurkan HTML dan JavaScript. 

<!-- Cara 1: Langsung tulis kode di atribut -->
<button onclick="alert('Halo!')">Klik</button>

<!-- Cara 2: Panggil fungsi yang sudah dibuat -->
<button onclick="inlineClick()">Klik Saya</button>

<!-- Cara 3: Kirim elemen dengan 'this' -->
<div onmouseover="inlineMouseOver(this)"
     onmouseout="inlineMouseOut(this)">
  Hover di sini
</div>
// Fungsi yang dipanggil dari atribut HTML
function inlineClick() {
  alert('Tombol diklik via inline handler!');
}

// Parameter 'element' menerima 'this' dari HTML
function inlineMouseOver(element) {
  element.style.boxShadow = '0 0 20px rgba(91,91,247,0.6)';
}

function inlineMouseOut(element) {
  element.style.boxShadow = 'none';
}

2. Event Handler Properties Menengah

Handler ditetapkan via properti DOM element, misal element.onclick = function() {...}. Lebih rapi dari inline, tapi hanya bisa satu handler per event. 

// Ambil elemen dari DOM
const btn = document.getElementById('btn-property');
const area = document.getElementById('property-area');

// Set handler via properti onclick
btn.onclick = function() {
  alert('Diklik via property!');
};

// Set handler via properti onmouseover & onmouseout
area.onmouseover = function() {
  this.style.borderColor = '#00ff88';
  this.textContent = 'Mouse ada di dalam!';
};

area.onmouseout = function() {
  this.style.borderColor = '#5b5bf7';
  this.textContent = 'Arahkan mouse ke sini';
};

// ⚠️ HATI-HATI: Properti hanya bisa 1 handler!
// Jika di-assign ulang, handler sebelumnya HILANG:
btn.onclick = function() {
  alert('Handler baru — yang lama sudah tergantikan!');
};

3. addEventListener() Method Direkomendasikan

Cara terbaik dan paling fleksibel. Bisa menambahkan banyak listener untuk satu event, mendukung opsi once, capture, dll. 

const btn = document.getElementById('myBtn');

// ✅ Bisa pasang BANYAK listener pada 1 event
btn.addEventListener('click', function() {
  console.log('Handler 1 jalan!');
});

btn.addEventListener('click', function() {
  console.log('Handler 2 juga jalan!');
});

// ✅ Opsi once: listener otomatis dihapus setelah 1x
btn.addEventListener('click', handleOnce, { once: true });

// ✅ Bisa hapus listener yang sudah terdaftar
function myHandler() { console.log('Hello'); }
btn.addEventListener('click', myHandler);
btn.removeEventListener('click', myHandler);

4. Mouse Events Interaktif

Semua event terkait mouse: click, dblclick, mouseover, mouseout, mousedown, mouseup, mousemove, dan contextmenu. 

const area = document.getElementById('mouse-area');

// click — klik biasa (kiri)
area.addEventListener('click', function(e) {
  console.log(`Klik di (${e.offsetX}, ${e.offsetY})`);
});

// dblclick — double-click
area.addEventListener('dblclick', function(e) {
  console.log('Double-click!');
});

// mouseover — mouse masuk elemen
area.addEventListener('mouseover', () => console.log('Mouse masuk'));

// mouseout — mouse keluar elemen
area.addEventListener('mouseout', () => console.log('Mouse keluar'));

// mousedown — tombol mouse ditekan
area.addEventListener('mousedown', function(e) {
  const btns = ['Kiri', 'Tengah', 'Kanan'];
  console.log(`Tombol ${btns[e.button]} ditekan`);
});

// mouseup — tombol mouse dilepas
area.addEventListener('mouseup', () => console.log('Tombol dilepas'));

// mousemove — mouse bergerak (sering dipanggil!)
area.addEventListener('mousemove', function(e) {
  tracker.style.left = e.offsetX + 'px';
  tracker.style.top  = e.offsetY + 'px';
});

// contextmenu — klik kanan, bisa cegah menu default
area.addEventListener('contextmenu', function(e) {
  e.preventDefault();  // cegah menu klik kanan
  console.log('Klik kanan dicegah!');
});

5. Keyboard Events Interaktif

Event keyboard: keydown (saat tombol ditekan), keyup (saat dilepas), dan keypress (deprecated). Menangkap informasi key, code, dan modifier keys. 

const input = document.getElementById('keyboard-area');

// keydown — dipanggil saat tombol DITEKAN
input.addEventListener('keydown', function(e) {
  console.log(`Key: ${e.key}`);   // 'a', 'Enter', 'Shift'
  console.log(`Code: ${e.code}`); // 'KeyA', 'Enter', 'ShiftLeft'

  // Cek modifier keys
  if (e.ctrlKey)  console.log('Ctrl ditekan');
  if (e.shiftKey) console.log('Shift ditekan');
  if (e.altKey)   console.log('Alt ditekan');

  // Contoh: cegah Ctrl+S
  if (e.ctrlKey && e.key === 's') {
    e.preventDefault();
    console.log('Ctrl+S dicegah!');
  }
});

// keyup — dipanggil saat tombol DILEPAS
input.addEventListener('keyup', function(e) {
  console.log(`Tombol ${e.key} dilepas`);
});

6. Form Events Penting

Event pada form: submit, reset, focus, blur, input (real-time), dan change (setelah selesai). 

const form = document.getElementById('demo-form');
const nameInput = document.getElementById('form-name');
const selectEl = document.getElementById('form-select');

// submit — saat form disubmit
form.addEventListener('submit', function(e) {
  e.preventDefault();  // PENTING! Cegah halaman refresh
  console.log('Nama:', nameInput.value);
});

// reset — saat form direset
form.addEventListener('reset', () => console.log('Form direset!'));

// focus — saat input mendapat fokus (diklik/tab)
nameInput.addEventListener('focus', () => console.log('Input difokuskan'));

// blur — saat input kehilangan fokus
nameInput.addEventListener('blur', () => console.log('Input blur'));

// input — setiap kali nilai berubah (real-time)
nameInput.addEventListener('input', function() {
  console.log('Mengetik:', this.value);
});

// change — SETELAH nilai berubah & kehilangan fokus
selectEl.addEventListener('change', function() {
  console.log('Pilihan:', this.value);
});

7. Window Events Global

Event pada object window: resize, scroll, online/offline, hashchange, dan load. Scroll indicator di atas halaman juga menggunakan scroll event! 

// load — halaman selesai dimuat (semua resource)
window.addEventListener('load', () => {
  console.log('Halaman sudah dimuat!');
});

// DOMContentLoaded — DOM siap (tanpa tunggu gambar, dll)
document.addEventListener('DOMContentLoaded', () => {
  console.log('DOM siap!');
});

// resize — ukuran window berubah
window.addEventListener('resize', () => {
  console.log(`Ukuran: ${window.innerWidth}x${window.innerHeight}`);
});

// scroll — halaman di-scroll
window.addEventListener('scroll', () => {
  const persen = (window.scrollY / 
    (document.documentElement.scrollHeight - window.innerHeight)) * 100;
  progressBar.style.width = persen + '%';
});

// online & offline — koneksi internet
window.addEventListener('online', () => alert('Kembali online!'));
window.addEventListener('offline', () => alert('Offline!'));

// beforeunload — sebelum halaman ditutup
window.addEventListener('beforeunload', function(e) {
  e.preventDefault();
  e.returnValue = ''; // Tampilkan dialog konfirmasi
});

8. Media Events Audio/Video

Event pada elemen <audio> dan <video>: play, pause, ended, timeupdate, volumechange, seeking, seeked, loadeddata. 

const audio = document.getElementById('media-player');

// play — media mulai diputar
audio.addEventListener('play', () => console.log('▶ Playing'));

// pause — media dijeda
audio.addEventListener('pause', () => console.log('⏸ Paused'));

// ended — media selesai diputar
audio.addEventListener('ended', () => console.log('⏹ Selesai'));

// timeupdate — posisi waktu berubah (sering!)
audio.addEventListener('timeupdate', function() {
  console.log(`Waktu: ${this.currentTime.toFixed(1)}s`);
});

// volumechange — volume berubah
audio.addEventListener('volumechange', function() {
  console.log(`Volume: ${Math.round(this.volume * 100)}%`);
});

// loadeddata — data media siap
audio.addEventListener('loadeddata', function() {
  console.log(`Durasi: ${this.duration.toFixed(1)}s`);
});

// seeking / seeked — user menggeser timeline
audio.addEventListener('seeking', () => console.log('Mencari...'));
audio.addEventListener('seeked', () => console.log('Posisi ditemukan'));

9. Touch Events Mobile

Event sentuh untuk perangkat mobile: touchstart, touchmove, touchend, dan touchcancel. Di desktop, mouse click akan mensimulasikan sentuhan. 

const area = document.getElementById('touch-area');

// touchstart — jari menyentuh layar
area.addEventListener('touchstart', function(e) {
  e.preventDefault();
  const touch = e.touches[0];  // sentuhan pertama
  console.log(`Sentuh di (${touch.clientX}, ${touch.clientY})`);
  console.log(`Total jari: ${e.touches.length}`);
});

// touchmove — jari bergerak di layar
area.addEventListener('touchmove', function(e) {
  e.preventDefault();
  const touch = e.touches[0];
  console.log(`Geser ke (${touch.clientX}, ${touch.clientY})`);
});

// touchend — jari diangkat dari layar
area.addEventListener('touchend', function(e) {
  console.log(`Jari dilepas, sisa: ${e.touches.length}`);
});

// touchcancel — sentuhan dibatalkan oleh sistem
area.addEventListener('touchcancel', () => {
  console.log('Sentuhan dibatalkan');
});

10. Drag and Drop Events Lanjutan

HTML5 Drag & Drop API: dragstart, dragend, dragover, dragenter, dragleave, dan drop.

🔹 10a. Seret Elemen Antar Zona (Kanban Board)

<!-- Item harus punya atribut draggable="true" -->
<div class="drag-item" id="drag-1" draggable="true">
  Belajar Event Handling
</div>

<!-- Zona drop target -->
<div class="drag-zone" id="zone-todo">
  <h4>📋 To Do</h4>
</div>
const item = document.getElementById('drag-1');
const zone = document.getElementById('zone-todo');

// dragstart — mulai menyeret
item.addEventListener('dragstart', function(e) {
  e.dataTransfer.setData('text/plain', this.id);
  e.dataTransfer.effectAllowed = 'move';
  this.classList.add('dragging');
});

// dragend — selesai menyeret
item.addEventListener('dragend', function() {
  this.classList.remove('dragging');
});

// dragover — WAJIB preventDefault agar drop bisa terjadi!
zone.addEventListener('dragover', function(e) {
  e.preventDefault();  // ← INI PENTING!
  e.dataTransfer.dropEffect = 'move';
});

// dragenter — item masuk zona drop
zone.addEventListener('dragenter', (e) => e.preventDefault());

// dragleave — item keluar zona drop
zone.addEventListener('dragleave', () => console.log('Keluar zona'));

// drop — item dijatuhkan di zona
zone.addEventListener('drop', function(e) {
  e.preventDefault();
  const id = e.dataTransfer.getData('text/plain');
  const el = document.getElementById(id);
  this.appendChild(el);  // pindahkan elemen
});

🔹 10b. Drag & Drop File dari Komputer

const dropArea = document.getElementById('file-drop-area');

// Cegah default agar browser tidak membuka file
['dragenter', 'dragover', 'dragleave', 'drop'].forEach(evt => {
  dropArea.addEventListener(evt, (e) => e.preventDefault());
});

// Tampilkan highlight saat file di-hover
dropArea.addEventListener('dragenter', () => {
  dropArea.classList.add('drag-over');
});
dropArea.addEventListener('dragleave', () => {
  dropArea.classList.remove('drag-over');
});

// drop — file dijatuhkan, baca datanya!
dropArea.addEventListener('drop', function(e) {
  const files = e.dataTransfer.files;  // FileList

  for (const file of files) {
    console.log(`Nama: ${file.name}`);
    console.log(`Tipe: ${file.type}`);
    console.log(`Ukuran: ${file.size} bytes`);

    // Preview gambar dengan FileReader
    if (file.type.startsWith('image/')) {
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = document.createElement('img');
        img.src = e.target.result;
        document.body.appendChild(img);
      };
      reader.readAsDataURL(file);
    }
  }
});

Seret file dari komputer (Explorer/Finder) dan jatuhkan ke area di bawah ini. File gambar akan ditampilkan preview-nya!
